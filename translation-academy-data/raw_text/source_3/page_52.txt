 ِطٝ سطظً جٌـحتٍز ئٌٝ ِـحٌ ٌِٚح: طرحقح أَ ِسحء ؟ 
 هٍ١ٕح ؾّ١وح أْ ٔطوحْٚ سٛ٠ح ٚجْ ٔىْٛ ِطسحِك١ٓ ِٓ جؾً أْ ٔو١ش ق١حز أفؼً 
 ً٘ ضوطمى أْ جٌّك١ؾ جٌٙحوٞ جورٍ قؿّح ِٓ جٌّك١ؾ جٌٕٙىٞ؟  
- سحفٍٔح ئٌٝ جٌسٛوجْ ٚلؼ١ٕح أسرٛه١ٓ ِن آنٝ جًٌٞ ٠وًّ ذحٌسفحٌز جٌّظٍ٠س ٕ٘حن. 
 ً٘ ضً٘د وجتّحُ ٌٍسرحقس ِن أطىلحته أغٕحء جٌوـٍس جٌظ١ف١س ؟  
هٕىِح جسط١ملص ذح لأِس ، وحْ أذٟ لى غحوٌ ئٌٝ ِىطرس. 
سًٕ٘د غىج ٌٍّسطشفٟ ق١ع سطؿٍٞ ٌؿىضٟ هٍّ١س ٌَم لٍد  
ً٘ ِٓ جٌسًٙ أْ ٠طسٍك جٌٍّء ؾرً ئفٍسص ذسٌٙٛس ً٘ز جلأ٠حَ ؟ 
 ضظىٌ ِظٍ جٌىػ١ٍ ِٓ جٌّٕطؿحش جٌؿ١ىز ٌٍهحٌؼ وً هحَ . 
ضسٍك جٌؿرحي أوػٍ طوٛذٗ ِٓ ٌوٛخ جٌىٌجؾحش ، أٌ١س وًجٌه؟ 
ضٕحٚي جٌىػ١ٍ ِٓ جٌسىٍ ٚجٌٍّف َجٌىْ٘ٛ ػحٌ ذحٌظكس ، أٌ١س وًٌه ؟ 
نى ٌجقطه فٝ جٌُّٕي ٚضفؼً أٜ ؿوحَ أٚ ِشٍٚخ سحنٓ ضٍ٠ىز. 
جٔظكه أْ ضمٛي جٌظىق وجتّح ،جلا ضىًخ أذىج . 
 ِشىٍٗ ضغ١ٍ جٌّٕحل ٚجٌضفحم وٌؾٗ قٍجٌز جلأٌع ضٙى٠ى ور١ٍ ٌٍر١ثٗ. 
 ٠ّٛش جٌىػ١ٍ ِٓ جلأفحٌلس ذسرد ٔمض جٌّ١حز ٚجٌـوحَ ٚجٔطشحٌ جلأٍِجع . 
 ٌمى ضكسٓ جلألطظحو جٌّظٍٞ ضكسٕحً  ور١ٍجً فٟ جلأهٛجَ جلأن١ٍز .  
ضمن ِى٠ٕٗ جٌغٍولٗ ٚ٘ٝ ِٕطؿن س١حقٝ ٌجتن ،هٍٝ شحؿة جٌركٍ جٌّطٛسؾ. 
غ١ٍ ِسّٛـ ٌسحتمٝ جٌس١حٌجش ذحلأٔطلحٌ ٕ٘ح ، لأْ ًٖ٘ جٌّٕـمٗ ِّٕٛم ف١ٙح جٔطلحٌ جٌس١حٌجش .  
 ؿحٌو ٌؾحي جٌشٍؿٗ جٌٍظٛص ٚضّىٕٛ فٝ جٌٕٙح٠ٗ ِٓ جٌمرغ هٍ١ُٙ .                                              
فٝ ٌأٜ جٌىػ١ٍ٠ٓ ، ٠وطرٍ جٌسفٍ ذحٌـحتٍز ِطوٗ ور١ٍز.                                                                
وً جٌّوٍِٛحش جٌطٝ ضٍ٠ى٘ح ِٛؾٛوز هٍٝ شرىٗ جٌّوٍِٛحش جٌىٌٚ١ٗ. 
لاضسطـ١ن جْ ضمٛو جٌس١حٌز وْٚ جٌكظٛي هٍٝ ٌنظٗ ل١حوٖ .                                                        
٠كًّ جلإٔسحْ ؾٍٛ٘ز غّ١ٕس فٟ ؾّؿّطٗ لا ٠ظىق أٔٙح ألٜٛ ِٓ أهلُ جٌكحسرحش ج٢ٌ١س: جٔٗ جٌّم.  
لا ضس١ٍ جٌٍّورحش فمؾ ذحٌرُٕ٠ٓ جٌكحٌٟ ِٓ جٌٍطحص ذً ضس١ٍ ذحٌغحَ جٌـر١وٟ وًٌه.   
ٌمى أوش جٌطكس١ٕحش فٟ نىِحش جٌظكس جٌوحِس ٚ جٌٍهح٠س جٌـر١س ئٌٝ ضمٍ١ً جٌٛف١حش ذ١ٓ جلأؿفحي جٌٍػن.  
جلأٔحٔ١ْٛ ٠فىٍْٚ وجتّح فٟ ِظٍكطُٙ جًٌجض١س ٚ لا ٠ُّٙٙ ِظٍكس ج٢نٍ٠ٓ.  
لا ٠ُجي ذحٌّؿطّن شٍفحء ٌى٠ُٙ ِٓ جٌسّوس جٌـ١رس ِح ٠ّىُٕٙ ِٓ هًّ ِح ٠مطٕوْٛ أٔٗ طٛجخ   
جٌؼّ١ٍ ٘ٛ ؿٛق جٌٕؿحز ٌكً ِشحوً ِظٍ ٚ جٌّظٍ٠١ٓ . 
لا شة ألرف ِٓ شحٌم ذلا أشؿحٌ فحٌهؼٍز ضؼفٟ ؾّحلا هٍ١ٗ.  
جلإٌجوز جٌمٛ٠س ضّىٓ جٌفٍو ِٓ ٔرً ٚ ضؿٕد جٌوحوجش جٌس١ثس ِػً جٌطىن١ٓ ٚ جلإسٍجف فٟ جٌّأوً ٚ جٌٍّرس . 
لا  ٠ٕىٍ أقى أْ جٌّحي ٠ّىٕٕح ِٓ شٍجء أش١حء وػ١ٍز ٚ ٌ١س وً شة فٙٛ ٌ١س جٌّظىٌ جٌٛق١ى ٌٍسوحوز.  
ٌٓ ضّف جٌوٌّٛس جٌػمحفس جٌهحطس ذىً شود . 
ٌمى أطرف جٌوٍُ ٠كىُ ٚ ٠طكىُ فٟ هحٌّٕح جٌكى٠ع فرٕحء جٌىٌٚس جٌكى٠ػس ٠وطّى أسحسح هٍٝ جٌوٍُ. 
وػ١ٍج ِح ضسّن جلأؿفحي ٠طّْٕٛ ٌٛ وحٔٛج ورحٌج ٚ ٠طّٕٝ جٌىرحٌ ٌٛ هحوٚج طغحٌج ٍِز أنٍٜ.  
ٕ٘حن جٌوى٠ى ِٓ جلأِحوٓ جٌٍجتوس جٌطٝ ٠ّىٕه َ٠حٌضٙح فٝ جٌمحٍ٘ز  ِػً جٌّطكف جٌّظٍٜ 
قٍطص جٌىٌٚس هٍٝ ئِىجو جٌّىجٌِ ذأؾُٙز جٌىّر١ٛضٍ ٌهىِس جٌطوٍ١ُ فٟ ِظٍ.  
لاشه أْ جٌٛقىز ذ١ٓ جلألـحٌ جٌوٍذ١س سٛف ضسحهى هٍٟ قً ِشىلاضٙح ضّحِح.
